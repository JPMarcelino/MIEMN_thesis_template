    {\normalfont\Large\bfseries}
    {\thesection}
    {1em}
    {}

\newglossarystyle{tabularabbr}{%
  \setglossarystyle{long}%
  \renewenvironment{theglossary}%
    {\begin{tabular}{p{0.18\linewidth} p{0.75\linewidth}}%
     \textbf{Abbreviation} & \textbf{Definition} \\ \hline}%
    {\end{tabular}}%
  \renewcommand*{\glossentry}[2]{%
    \glsentryshort{##1} & \glsentrylong{##1} \\%
  }%
}
\newglossarystyle{tabularnotation}{%
  \setglossarystyle{long}%
  \renewenvironment{theglossary}%
    {\begin{tabular}{p{0.25\linewidth} p{0.75\linewidth}}%
     \textbf{Symbol} & \textbf{Meaning} \\ \hline}%
    {\end{tabular}}%
  \renewcommand*{\glossentry}[2]{%
    \glsentryname{##1} & \glossentrydesc{##1} \\%
  }%
}

% Include a file from sections/ : \importsection{introduction}
\newcommand{\importsection}[1]{\input{sections/#1}}

% Unnumbered front-matter heading AND a ToC entry.
\newcommand{\frontchapter}[1]{%
  \cleardoublepage
  \chapter*{#1}%
  \phantomsection
  \addcontentsline{toc}{chapter}{#1}%
}

% Title page, driven entirely by metadata.tex.
\newcommand{\frontpage}{%
  \begin{titlepage}
    \centering
    \includegraphics[width=0.3\textwidth]{university-logo.png}\par
    \vspace{2cm}
    {\Huge\bfseries \thesistitle\par}
    \vspace{1.5cm}
    {\Large\bfseries \thesisauthor\par}
    \vfill
    A dissertation submitted in fulfillment of the requirements\\
    for the degree of \emph{\thesisdegree}\\
    in \thesisprogram\par
    \vspace{2cm}
    {\bfseries Supervisors:}\\[0.5em]
    \thesissupervisors\par
    \vspace{2cm}
    {\bfseries Examining Committee:}\\[0.5em]
    \thesiscommittee\par
    \vspace{2cm}
    \thesisuniversity\\
    \thesisschool\par
    \vspace{1cm}
    {\large \thesisdate\par}
  \end{titlepage}
}

% Mandatory copyright page (content in sections/copyright.tex).
\newcommand{\copyrightpage}{%
  \thispagestyle{empty}%
  \vspace*{0.1\textheight}%
  \importsection{copyright}%
  \cleardoublepage
}
